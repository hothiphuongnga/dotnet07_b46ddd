- Tạo sln 
    - 4 projects: Domain, Infrastructure, Application, Presentation

    - src
        - SalesDb.Domain
        - SalesDb.Infrastructure (repository) 
        - SalesDb.Application
        - SalesDb.Presentation
    SalesDb.sln





    STEP
    dotnet new sln -n SalesDb
    dotnet new classlib -n SalesDb.Domain -o src/SalesDb.Domain 
    dotnet new classlib -n SalesDb.Application -o src/SalesDb.Application 
    dotnet new classlib -n SalesDb.Infrastructure -o src/SalesDb.Infrastructure 
    dotnet new webapi -n SalesDb.Api -o src/SalesDb.Api

    dotnet sln add src/**/*.csproj

    -- thêm phụ thuộc
    dotnet add src/SalesDb.Application reference src/SalesDb.Domain
    dotnet add src/SalesDb.Infrastructure reference src/SalesDb.Application
    dotnet add src/SalesDb.Api reference src/SalesDb.Application src/SalesDb.Infrastructure

    -- Cai nuget    
    # infrastructure
    dotnet add src/SalesDb.Infrastructure package Microsoft.EntityFrameworkCore.SqlServer
    dotnet add src/SalesDb.Infrastructure package Microsoft.EntityFrameworkCore.Design
    dotnet add src/SalesDb.Infrastructure package Microsoft.Extensions.Caching.StackExchangeRedis
    dotnet add src/SalesDb.Infrastructure package System.IdentityModel.Tokens.Jwt

    # Presentation
    dotnet add src/SalesDb.Api package Microsoft.AspNetCore.Authentication.JwtBearer
    dotnet add src/SalesDb.Api package Swashbuckle.AspNetCore

    # application
    dotnet add src/SalesDb.Application package AutoMapper
